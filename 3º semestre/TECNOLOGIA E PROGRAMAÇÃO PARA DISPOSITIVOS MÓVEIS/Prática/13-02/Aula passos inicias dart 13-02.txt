String veredito (int nota){
  if (nota >= 5) return 'Aprovado';
  return 'Reprovado';
}

void main(){
  int? nota = null;
  if (nota == null) print ('Não tenho como te dar um veredito, pois não consegui obter sua nota');
  else print(veredito(nota));
}
